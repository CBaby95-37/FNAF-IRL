const peer = new Peer(); let activeStreams = {}; let dummyStream = null; // Create a blank "dummy" stream so the monitor can "call" the cameras function createDummyStream() { const canvas = document.createElement('canvas'); canvas.width = 10; canvas.height = 10; const stream = canvas.captureStream(1); return stream; } peer.on('open', (id) => { document.getElementById('my-id').innerText = id; dummyStream = createDummyStream(); // Initialize dummy stream }); // --- CAMERA CODE (The Phones) --- async function startCamera() { try { const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "environment" }, audio: false }); // When the monitor "calls" this phone, answer with the phone's camera video peer.on('call', (call) => { call.answer(stream); }); alert("CAMERA ACTIVE. Keep this screen open."); } catch (err) { alert("Camera error: " + err.message); } } // --- MONITOR CODE (The PC/Laptop) --- function connectToCamera() { const remoteId = document.getElementById('peer-id-input').value.trim(); if (!remoteId) { alert("Please enter a Camera ID first!"); return; } if (!peer || peer.destroyed) { alert("Peer connection lost. Refresh the page."); return; } // CRITICAL FIX: We must send a stream (dummyStream) to the phone // so the phone can "answer" back with its camera video. const call = peer.call(remoteId, dummyStream); if (!call) { alert("Failed to create call object. Is the ID correct?"); return; } call.on('stream', (remoteStream) => { const camNum = Object.keys(activeStreams).length + 1; const camLabel = `CAM ${camNum}`; activeStreams[camLabel] = remoteStream; addButton(camLabel); document.getElementById('peer-id-input').value = ""; alert(`Successfully connected to ${camLabel}`); }); call.on('error', (err) => { alert("Connection error: " + err.type); }); } function addButton(label) { const btn = document.createElement('button'); btn.className = 'cam-btn'; btn.innerText = label; btn.onclick = () => switchCam(label); document.getElementById('btn-container').appendChild(btn); } function switchCam(label) { const video = document.getElementById('video-feed'); document.getElementById('cam-label').innerText = label; if (activeStreams[label]) { video.srcObject = activeStreams[label]; video.play().catch(e => console.error("Playback failed", e)); // Visual Glitch Effect document.getElementById('static').style.opacity = "0.6"; setTimeout(() => { document.getElementById('static').style.opacity = "0.1"; }, 400); } } function showMonitor() { if (Object.keys(activeStreams).length === 0) { alert("Add at least one camera first!"); return; } document.getElementById('setup').style.display = 'none'; document.getElementById('monitor-ui').style.display = 'grid'; const firstCam = Object.keys(activeStreams)[0]; switchCam(firstCam); }
